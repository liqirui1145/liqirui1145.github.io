{{- $dest := .Destination -}} {{- $alt := .PlainText | default .Title -}} {{- $img := .Page.Resources.GetMatch $dest -}} {{- if not $img -}}{{- $img = resources.Get $dest -}}{{- end -}} {{- $isRaster := false -}} {{- with $img -}} {{- if in (slice "jpeg" "png" "webp") .MediaType.SubType -}}{{- $isRaster = true -}}{{- end -}} {{- end -}} {{- if and $img $isRaster -}} {{- $w := $img.Width -}} {{- $h := $img.Height -}} {{- $widths := slice 480 800 1200 -}} {{- $srcset := slice -}} {{- range $target := $widths -}} {{- if le $target $w -}} {{- $r := $img.Resize (printf "%dx webp q82" $target) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $target) -}} {{- end -}} {{- end -}} {{- if not (in $widths $w) -}} {{- $full := $img.Resize (printf "%dx webp q82" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" $full.RelPermalink $w) -}} {{- end -}} {{- else -}} {{- end -}}
